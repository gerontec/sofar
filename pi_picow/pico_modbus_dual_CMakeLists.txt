# Raspberry Pi Pico Modbus Dual-Core Controller
# Build with Pico SDK

cmake_minimum_required(VERSION 3.13)

# Pico SDK setup (adjust path as needed)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)

project(pico_modbus_dual C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

# Initialize Pico SDK
pico_sdk_init()

# Main executable
add_executable(pico_modbus_dual
    pico_modbus_dual.c
)

# Link libraries
target_link_libraries(pico_modbus_dual
    pico_stdlib          # Standard library
    pico_multicore       # Dual-core support
    pico_sync            # Mutex/semaphore
    hardware_uart        # UART hardware
    hardware_gpio        # GPIO hardware
    hardware_timer       # Timer hardware
    hardware_watchdog    # Watchdog hardware
)

# Enable USB serial output
pico_enable_stdio_usb(pico_modbus_dual 1)
pico_enable_stdio_uart(pico_modbus_dual 0)

# Create UF2 file for drag-and-drop programming
pico_add_extra_outputs(pico_modbus_dual)

# Compile options
target_compile_options(pico_modbus_dual PRIVATE
    -Wall
    -Wextra
    -Werror=implicit-function-declaration
    -O2  # Optimize for speed
)
